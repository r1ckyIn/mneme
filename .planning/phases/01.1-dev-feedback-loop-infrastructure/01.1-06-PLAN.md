// ----- Pure formatters — testable without Tauri runtime -----

pub(crate) fn iso8601_now() -> String {
    // Lightweight ISO8601 without chrono dep. Format: YYYY-MM-DDTHH:MM:SSZ.
    let secs = SystemTime::now().duration_since(UNIX_EPOCH).unwrap_or_default().as_secs();
    let (days, time_of_day) = (secs / 86_400, secs % 86_400);
    let (mut y, mut d) = (1970i64, days as i64);
    loop {
        let yd = if is_leap(y) { 366 } else { 365 };
        if d < yd { break; }
        d -= yd; y += 1;
    }
    let mdays = if is_leap(y) {
        [31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
    } else {
        [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31]
    };
    let mut m = 1; let mut day_in_month = d + 1;
    for &days_in_m in &mdays {
        if day_in_month <= days_in_m { break; }
        day_in_month -= days_in_m; m += 1;
    }
    let h = time_of_day / 3600; let min = (time_of_day % 3600) / 60; let s = time_of_day % 60;
    format!("{:04}-{:02}-{:02}T{:02}:{:02}:{:02}Z", y, m, day_in_month, h, min, s)
}

fn is_leap(y: i64) -> bool { (y % 4 == 0 && y % 100 != 0) || y % 400 == 0 }

pub(crate) fn format_console_entry(level: &str, message: &str, tag: Option<&str>, source: Option<&str>, line: Option<u32>) -> String {
    let ts = iso8601_now();
    let tag_str = tag.unwrap_or(level).to_uppercase();
    let source_str = source.map(|s| format!("{s}:{}", line.unwrap_or(0))).unwrap_or_default();
    format!("[FRONTEND_CONSOLE]{}|{}|{}|{}\n", tag_str, ts, message, source_str)
}

pub(crate) fn format_network_entry(method: &str, url: &str, status: u16, duration_ms: f64) -> String {
    let ts = iso8601_now();
    format!("[FRONTEND_NETWORK]{}|{}|{}|{}|{}ms\n", method, ts, url, status, duration_ms)
}

pub(crate) fn format_perf_entry(metric: &str, value: f64, ts: &str) -> String {
    format!("[FRONTEND_PERF]{}|{}|{}\n", metric, ts, value)
}

use std::path::PathBuf;
use tokio::sync::mpsc::{self, Sender};
use tokio::time::{sleep, Duration};

const ROTATION_THRESHOLD_BYTES: u64 = 10 * 1024 * 1024;
const DEBOUNCE_MS: u64 = 100;
const CHANNEL_BUFFER: usize = 1024;

#[derive(Clone, Copy, Debug)]
pub enum LogChannel { Console, Network, Perf }

impl LogChannel {
    fn filename(&self) -> &'static str {
        match self { Self::Console => "console.log", Self::Network => "network.log", Self::Perf => "perf.log" }
    }
}

/// Start a dedicated writer task for one log channel.
///
/// Returns a Sender<String> the Tauri command handlers push into. Background
/// task batches with 100 ms debounce and rotates at 10 MB per D-SF-03. All IO
/// errors swallowed per D-SF-04 — the dev loop MUST NOT crash on log-write
/// failure.
pub fn start_log_writer(_channel: LogChannel, path: PathBuf) -> Sender<String> {
    let (tx, mut rx) = mpsc::channel::<String>(CHANNEL_BUFFER);
    tokio::spawn(async move {
        let mut buf: Vec<String> = Vec::with_capacity(64);
        loop {
            tokio::select! {
                msg = rx.recv() => {
                    match msg {
                        Some(s) => buf.push(s),
                        None => { flush(&path, &mut buf); break; }
                    }
                }
                _ = sleep(Duration::from_millis(DEBOUNCE_MS)) => {
                    if !buf.is_empty() { flush(&path, &mut buf); }
                }
            }
            if buf.len() >= 256 { flush(&path, &mut buf); }
        }
    });
    tx
}

fn flush(path: &PathBuf, buf: &mut Vec<String>) {
    use std::fs::OpenOptions;
    use std::io::Write;
    if buf.is_empty() { return; }
    if let Some(parent) = path.parent() { let _ = std::fs::create_dir_all(parent); }

    let payload_len: u64 = buf.iter().map(|s| s.len() as u64).sum();
    let current_size = std::fs::metadata(path).map(|m| m.len()).unwrap_or(0);
    if current_size + payload_len > ROTATION_THRESHOLD_BYTES {
        let backup = path.with_extension("log.1");
        let _ = std::fs::rename(path, &backup);
    }

    let mut file = match OpenOptions::new().append(true).create(true).open(path) {
        Ok(f) => f,
        Err(_) => { buf.clear(); return; }  // D-SF-04 swallow
    };
    for line in buf.drain(..) {
        let _ = file.write_all(line.as_bytes());
    }
    let _ = file.flush();
}

pub struct DevWriter {
    pub console_tx: tokio::sync::Mutex<Option<Sender<String>>>,
    pub network_tx: tokio::sync::Mutex<Option<Sender<String>>>,
    pub perf_tx: tokio::sync::Mutex<Option<Sender<String>>>,
    pub log_dir: tokio::sync::Mutex<Option<PathBuf>>,
}

impl DevWriter {
    pub fn new(log_dir: PathBuf) -> Self {
        Self {
            console_tx: tokio::sync::Mutex::new(None),
            network_tx: tokio::sync::Mutex::new(None),
            perf_tx: tokio::sync::Mutex::new(None),
            log_dir: tokio::sync::Mutex::new(Some(log_dir)),
        }
    }
}

impl Default for DevWriter {
    fn default() -> Self {
        let dir = std::env::current_dir().unwrap_or_default().join(".dev-logs");
        Self::new(dir)
    }
}

async fn ensure_channel(state: &State<'_, DevWriter>, ch: LogChannel) -> Result<(), String> {
    let log_dir_g = state.log_dir.lock().await;
    let log_dir = log_dir_g.as_ref().ok_or("log_dir not set")?.clone();
    drop(log_dir_g);
    let path = log_dir.join(ch.filename());

    let slot: &tokio::sync::Mutex<Option<Sender<String>>> = match ch {
        LogChannel::Console => &state.console_tx,
        LogChannel::Network => &state.network_tx,
        LogChannel::Perf    => &state.perf_tx,
    };
    let mut guard = slot.lock().await;
    if guard.is_none() {
        *guard = Some(start_log_writer(ch, path));
    }
    Ok(())
}

// ----- dev_capture_screenshot (E1 — screencapture primary) -----

#[tauri::command]
pub async fn dev_capture_screenshot<R: Runtime>(
    app: AppHandle<R>,
    scope: String,
) -> Result<String, String> {
    use std::process::Command;
    let cwd = std::env::current_dir().map_err(|e| e.to_string())?;
    let log_dir = cwd.join(".dev-logs/screenshots");
    std::fs::create_dir_all(&log_dir).map_err(|e| e.to_string())?;
    let ts = iso8601_now().replace(':', "-");
    let out_path = log_dir.join(format!("{ts}.png"));

    let result = match scope.as_str() {
        "webview" | "Webview" => {
            let main = app.get_webview_window("main").ok_or("main window not found")?;
            let window_id = resolve_cg_window_id(&main)?;
            Command::new("/usr/sbin/screencapture")
                .arg("-l").arg(window_id.to_string())
                .arg("-x").arg("-o").arg(&out_path)
                .status().map_err(|e| e.to_string())?
        }
        "window" | "Window" => {
            let main = app.get_webview_window("main").ok_or("main window not found")?;
            let pos = main.outer_position().map_err(|e| e.to_string())?;
            let size = main.outer_size().map_err(|e| e.to_string())?;
            let rect = format!("{},{},{},{}", pos.x, pos.y, size.width, size.height);
            Command::new("/usr/sbin/screencapture")
                .arg("-R").arg(rect)
                .arg("-x").arg(&out_path)
                .status().map_err(|e| e.to_string())?
        }
        other => return Err(format!("unknown scope: {other}")),
    };
    if !result.success() { return Err(format!("screencapture exited with {result}")); }
    out_path.to_str().ok_or_else(|| "path not utf8".to_string()).map(|s| s.to_string())
}

#[cfg(target_os = "macos")]
fn resolve_cg_window_id<R: Runtime>(window: &tauri::WebviewWindow<R>) -> Result<i64, String> {
    use objc2::msg_send;
    use objc2_app_kit::NSWindow;
    let ns_ptr = window.ns_window().map_err(|e| e.to_string())? as *mut NSWindow;
    // SAFETY: ns_window() returns a valid NSWindow pointer for the lifetime of the window.
    // windowNumber returns NSInteger (i64 on Mac); safe cast.
    let number: i64 = unsafe { msg_send![ns_ptr, windowNumber] };
    Ok(number)
}

// src-tauri/src/bin/dev_invoke.rs — D-TR-03 + E2 errata: dev-only CLI for
// invoking registered Tauri commands from outside the running app.
//
// Build: cargo build --bin dev-invoke --features dev-invoke
// Run:   target/debug/dev-invoke <command> [args...]
//
// Examples:
//   target/debug/dev-invoke dev_capture_screenshot webview
//   target/debug/dev-invoke dev_query_state
//
// The npm-script bridge (plan 01.1-07) shells out to this binary so the SDK
// can reach Tauri commands without an in-app round-trip. The binary
// constructs a minimal Tauri runtime context, invokes the command, prints
// the result to stdout, and exits.

#![cfg(debug_assertions)]

use std::env;
use std::process::ExitCode;

fn main() -> ExitCode {
    let args: Vec<String> = env::args().collect();
    if args.len() < 2 {
        eprintln!(r#"{{"error":"usage: dev-invoke <command> [args...]","commands":["dev_capture_screenshot","dev_query_state","dev_log_console_entry","dev_log_network_entry","dev_log_perf_entry"]}}"#);
        return ExitCode::from(2);
    }

    let cmd = &args[1];
    let rest: Vec<String> = args.iter().skip(2).cloned().collect();

    // For commands that need a running webview (screenshot / query_state),
    // we cannot fabricate one in this CLI — the binary is meant to be invoked
    // while `cargo tauri dev` is also running. The CLI's job is to expose the
    // command surface; the implementation in src/dev.rs may need to operate
    // on filesystem (.dev-logs paths) directly OR via a side-channel to the
    // running app.
    //
    // v1 pragmatic choice: dev-invoke shells out to the same syscalls the
    // commands would have run (screencapture, file appends). The running
    // `cargo tauri dev` instance is what actually maintains the writer task
    // for log appends — but log entries are also useful when invoked from
    // dev-invoke (they tee to .dev-logs/* directly without going through
    // Tauri's invoke handler). This is acceptable because both writers use
    // append + 10MB rotation; the inevitable concurrent-append on shared
    // file is safe under POSIX append semantics.

    let result: Result<String, String> = match cmd.as_str() {
        "dev_capture_screenshot" => {
            let scope = rest.first().cloned().unwrap_or_else(|| "webview".to_string());
            cli_capture_screenshot(&scope)
        }
        "dev_query_state" => {
            // Without a running webview accessible from this fresh process, we
            // signal the running app to dispatch via filesystem touchfile.
            // Caller should tail .dev-logs/console.log for [snapshot]<json>.
            std::fs::create_dir_all(".dev-logs").map_err(|e| e.to_string())?;
            std::fs::write(".dev-logs/snapshot.request", "1").map_err(|e| e.to_string())?;
            Ok(r#"{"hint":"signal written; tail .dev-logs/console.log for [snapshot] line"}"#.to_string())
        }
        "dev_log_console_entry" | "dev_log_network_entry" | "dev_log_perf_entry" => {
            // Direct filesystem append (bypasses running app's writer task).
            // Acceptable since POSIX append-mode writes are atomic.
            Err(format!("{cmd} via CLI not v1 — use Tauri runtime"))
        }
        other => Err(format!("unknown command: {other}")),
    };

    match result {
        Ok(s) => { println!("{s}"); ExitCode::SUCCESS }
        Err(e) => { eprintln!(r#"{{"error":"{}"}}"#, e.replace('"', "\\\"")); ExitCode::from(1) }
    }
}

fn cli_capture_screenshot(scope: &str) -> Result<String, String> {
    use std::process::Command;
    let log_dir = std::env::current_dir().map_err(|e| e.to_string())?.join(".dev-logs/screenshots");
    std::fs::create_dir_all(&log_dir).map_err(|e| e.to_string())?;
    let ts = app_lib::dev::iso8601_now().replace(':', "-");
    let out_path = log_dir.join(format!("{ts}.png"));

    // For CLI-mode webview capture, we need the CGWindowID of the currently-running
    // `mneme` Tauri app. We use osascript as a portable fallback to query the
    // frontmost window of the mneme process. Acceptable for v1 — Phase 01.2 UDS
    // upgrade per RESEARCH spike can replace this with a hot-path IPC call.

    let status = match scope {
        "webview" | "window" => {
            // Use osascript to get frontmost window id of mneme process
            let osa_script = r#"tell application "System Events"
              tell process "mneme"
                return id of window 1
              end tell
            end tell"#;
            let win_id_output = Command::new("osascript")
                .arg("-e").arg(osa_script)
                .output().map_err(|e| format!("osascript not available: {e}"))?;
            if !win_id_output.status.success() {
                return Err(format!("osascript failed: {}", String::from_utf8_lossy(&win_id_output.stderr)));
            }
            let win_id = String::from_utf8_lossy(&win_id_output.stdout).trim().to_string();
            Command::new("/usr/sbin/screencapture")
                .arg("-l").arg(&win_id)
                .arg("-x").arg("-o").arg(&out_path)
                .status().map_err(|e| e.to_string())?
        }
        other => return Err(format!("unknown scope: {other}")),
    };
    if !status.success() { return Err(format!("screencapture exited with {status}")); }
    out_path.to_str().ok_or_else(|| "path not utf8".to_string()).map(|s| s.to_string())
}
