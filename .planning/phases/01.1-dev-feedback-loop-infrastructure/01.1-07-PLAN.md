```javascript
#!/usr/bin/env node
// scripts/gsd-dev-snapshot.mjs — D-BR-01 + D-BR-02 npm-script bridge.
//
// Triggers dev_query_state which writes a [snapshot]<json> line into
// .dev-logs/console.log; then tails the log for the line and prints the JSON
// payload to stdout. v1 approach per plan 01.1-06 (Phase 01.2 UDS upgrade
// replaces tail with direct round-trip).

import { existsSync, readFileSync, statSync } from 'node:fs';
import { spawnSync } from 'node:child_process';
import { resolve } from 'node:path';

const pidPath = resolve(process.cwd(), '.dev-logs', 'dev-server.pid');
if (!existsSync(pidPath)) {
  process.stderr.write(JSON.stringify({ error: 'dev server not running', hint: 'run gsd-sdk query verify.start-dev-loop first' }) + '\n');
  process.exit(1);
}

const consoleLog = resolve(process.cwd(), '.dev-logs', 'console.log');
const startSize = existsSync(consoleLog) ? statSync(consoleLog).size : 0;
const startTs = Date.now();

const devInvoke = resolve(process.cwd(), 'src-tauri', 'target', 'debug', 'dev-invoke');
if (!existsSync(devInvoke)) {
  process.stderr.write(JSON.stringify({ error: 'dev-invoke binary missing', hint: 'run cargo build --bin dev-invoke --features dev-invoke from src-tauri/' }) + '\n');
  process.exit(1);
}

// Trigger the snapshot
const trigger = spawnSync(devInvoke, ['dev_query_state'], { encoding: 'utf8' });
if (trigger.status !== 0) {
  process.stderr.write(trigger.stderr || JSON.stringify({ error: 'dev-invoke dev_query_state failed' }));
  process.exit(trigger.status || 1);
}

// Poll console.log for new [snapshot] line — give the running app up to 3s to write
const POLL_DEADLINE_MS = 3000;
const POLL_INTERVAL_MS = 80;
let snapshot = null;
while (Date.now() - startTs < POLL_DEADLINE_MS) {
  if (existsSync(consoleLog)) {
    const newSize = statSync(consoleLog).size;
    if (newSize > startSize) {
      const tail = readFileSync(consoleLog, 'utf8').slice(startSize);
      // Match the most recent [snapshot]<json> line
      const m = tail.match(/\[snapshot\](\{[\s\S]*?\})\s*$/m);
      if (m) { snapshot = m[1]; break; }
    }
  }
  // Busy-wait — avoid setTimeout to keep this fully synchronous for npm script use
  const deadline = Date.now() + POLL_INTERVAL_MS;
  while (Date.now() < deadline) { /* spin */ }
}

if (!snapshot) {
  process.stderr.write(JSON.stringify({ error: 'timeout waiting for snapshot', hint: '__mnemeDevSnapshot__ may not be registered in the running webview' }) + '\n');
  process.exit(1);
}

process.stdout.write(snapshot + '\n');
```

#!/usr/bin/env node
// scripts/gsd-dev-scan-logs.mjs — D-BR-01 + D-BR-02 npm-script bridge.
//
// Greps .dev-logs/{console,network,perf}.log + .dev-logs/tauri.log for entries
// newer than --since (ISO 8601 timestamp). Prints structured JSON to stdout
// per the contract documented in plan 01.1-04 verify.scan-signals consumer.
// Exits 1 with structured stderr JSON on dev-server-not-running per D-BR-02.

import { existsSync, readFileSync } from 'node:fs';
import { resolve, join } from 'node:path';

const argv = process.argv.slice(2);
let since = '';
for (let i = 0; i < argv.length; i += 1) {
  if (argv[i] === '--since' && argv[i + 1]) { since = argv[i + 1]; i += 1; }
  else if (argv[i].startsWith('--since=')) { since = argv[i].split('=')[1]; }
}

// Validate --since is parseable ISO 8601 if provided
let sinceMs = 0;
if (since) {
  sinceMs = Date.parse(since);
  if (Number.isNaN(sinceMs)) {
    process.stderr.write(JSON.stringify({ error: `invalid --since: ${since}`, hint: 'use ISO 8601 like 2026-05-12T00:00:00Z' }) + '\n');
    process.exit(1);
  }
}

// D-BR-02 — fail loud if dev server not running
const pidPath = resolve(process.cwd(), '.dev-logs', 'dev-server.pid');
if (!existsSync(pidPath)) {
  process.stderr.write(JSON.stringify({ error: 'dev server not running', hint: 'run gsd-sdk query verify.start-dev-loop first' }) + '\n');
  process.exit(1);
}

const devLogsDir = resolve(process.cwd(), '.dev-logs');
const issues = [];
const summary = { console_errors: 0, console_warnings: 0, network_failures: 0, perf_concerns: 0 };

// Parse console.log lines: [FRONTEND_CONSOLE]TAG|<ts>|<message>|<source>:<line>|
function parseConsoleLine(line) {
  const m = line.match(/^\[FRONTEND_CONSOLE\]([A-Z\-]+)\|([^|]+)\|(.*?)\|(.*?)(?::(\d+))?\|?$/);
  if (!m) return null;
  const tag = m[1].toLowerCase();
  const ts = m[2];
  const message = m[3];
  const source = m[4];
  const line_no = m[5] ? Number(m[5]) : null;
  const level = ['log', 'info', 'debug', 'warn', 'error'].includes(tag) ? tag : 'error';
  return { level, tag, ts, message, source, line: line_no };
}

// Parse network.log lines: [FRONTEND_NETWORK]METHOD|<ts>|<url>|<status>|<dur>ms
function parseNetworkLine(line) {
  const m = line.match(/^\[FRONTEND_NETWORK\]([A-Z]+)\|([^|]+)\|([^|]+)\|(\d+)\|([\d.]+)ms$/);
  if (!m) return null;
  return { method: m[1], ts: m[2], url: m[3], status: Number(m[4]), duration_ms: Number(m[5]) };
}

// Parse perf.log lines: [FRONTEND_PERF]METRIC|<ts>|<value>
function parsePerfLine(line) {
  const m = line.match(/^\[FRONTEND_PERF\]([^|]+)\|([^|]+)\|([\d.]+)$/);
  if (!m) return null;
  return { metric: m[1], ts: m[2], value: Number(m[3]) };
}

function entryIsAfterSince(ts) {
  if (!sinceMs) return true;
  const entryMs = Date.parse(ts);
  if (Number.isNaN(entryMs)) return true;
  return entryMs >= sinceMs;
}

function scanFile(filename, parser, classifyFn) {
  const path = join(devLogsDir, filename);
  if (!existsSync(path)) return;
  const lines = readFileSync(path, 'utf8').split('\n').filter(Boolean);
  for (const line of lines) {
    const parsed = parser(line);
    if (parsed && entryIsAfterSince(parsed.ts)) {
      issues.push({ ...parsed, log_file: filename });
      classifyFn(parsed);
    }
  }
}

scanFile('console.log', parseConsoleLine, (e) => {
  if (e.level === 'error' || e.tag === 'uncaught' || e.tag === 'unhandled-rejection' || e.tag === 'resource-error') summary.console_errors += 1;
  else if (e.level === 'warn' || e.tag === 'csp') summary.console_warnings += 1;
});
scanFile('network.log', parseNetworkLine, (e) => {
  if (e.status === 0 || e.status >= 400) summary.network_failures += 1;
});
scanFile('perf.log', parsePerfLine, (e) => {
  if ((e.metric === 'largest-contentful-paint' && e.value > 2500) ||
      (e.metric === 'longtask' && e.value > 50)) {
    summary.perf_concerns += 1;
  }
});

// tauri.log is unstructured; include as fallback issues if present
const tauriLogPath = join(devLogsDir, 'tauri.log');
if (existsSync(tauriLogPath)) {
  const lines = readFileSync(tauriLogPath, 'utf8').split('\n').filter(Boolean);
  for (const line of lines) {
    if (/error|panic|warning/i.test(line)) {
      issues.push({ raw: line, log_file: 'tauri.log' });
    }
  }
}

process.stdout.write(JSON.stringify({ issues, summary, since: since || null }) + '\n');
