```typescript
const FORBIDDEN_BUCKETS = ['hybrid','terminal','console','dom-check','log-paste','command-run'];
const FORBIDDEN_PHRASINGS = [

function parseArg(args: string[], name: string): string | undefined {
  for (const a of args) {
    if (a.startsWith(`--${name}=`)) return a.slice(name.length + 3);
    if (a === `--${name}`) return args[args.indexOf(a) + 1];
  }
  if (name === 'phase' && args[0] && !args[0].startsWith('-')) return args[0];
  return undefined;
}

export const verifyValidateHtml: QueryHandler = async (args) => {
  const path = parseArg(args, 'path');
  if (!path) return { data: { error: 'path required', hint: 'pass --path <file>' } };
  if (!existsSync(path)) return { data: { error: 'file_not_found', path } };
  const content = readFileSync(path, 'utf8');
  const matches = [...content.matchAll(/<section\b[^>]*\bdata-bucket=["']([^"']+)["']/gi)];
  if (matches.length === 0) return { data: { error: 'no_sections_found', hint: 'template malformed', path } };
  const buckets = matches.map((m) => m[1]);
  const forbidden = buckets.filter((b) => FORBIDDEN_BUCKETS.includes(b));
  if (forbidden.length) return { data: { error: 'forbidden_bucket', found: [...new Set(forbidden)].sort(), allowed: [...ALLOWED_BUCKETS], path } };
  const unknown = buckets.filter((b) => !(ALLOWED_BUCKETS as readonly string[]).includes(b));
  if (unknown.length) return { data: { error: 'unknown_bucket', found: [...new Set(unknown)].sort(), allowed: [...ALLOWED_BUCKETS], path } };
  const phrasings: string[] = [];
  for (const re of FORBIDDEN_PHRASINGS) { const m = content.match(re); if (m) phrasings.push(m[0]); }
  if (phrasings.length) return { data: { error: 'forbidden_phrasing', matches: phrasings, path } };
  return { data: { passed: true, bucket_count: buckets.length, buckets_found: buckets, path } };
};
```

```typescript
export const verifyStartDevLoop: QueryHandler = async (args, projectDir) => {
  const phase = parseArg(args, 'phase') || (args[0]?.startsWith('-') ? undefined : args[0]);
  if (!phase) return { data: { error: 'phase required', hint: 'pass <phase> or --phase=<N>' } };
  const surface = parseArg(args, 'surface') || 'chromium';
  const devLogsDir = join(projectDir, '.dev-logs');
  mkdirSync(devLogsDir, { recursive: true });
  const probe = spawnSync('lsof', ['-i', `:${VITE_PORT}`], { encoding: 'utf8' });
  if (probe.status === 0 && probe.stdout.trim()) {
    const lines = probe.stdout.trim().split('\n').filter((l) => !/^COMMAND\s/.test(l));
    const pidMatch = lines[0]?.match(/\S+\s+(\d+)/);
    return { data: { error: `port ${VITE_PORT} in use`, existing_pid: pidMatch ? Number(pidMatch[1]) : null, hint: 'run verify.stop-dev-loop first' } };
  }
  const cmd = surface === 'tauri' ? 'cargo' : 'npm';
  const cmdArgs = surface === 'tauri' ? ['tauri', 'dev'] : ['run', 'dev'];
  const child = spawn(cmd, cmdArgs, { cwd: projectDir, detached: true, stdio: 'ignore' });
  child.unref();
  writeFileSync(join(devLogsDir, 'dev-server.pid'), String(child.pid ?? ''));
  writeFileSync(join(devLogsDir, 'dev-server.port'), String(VITE_PORT));
  return { data: { phase, surface, pid: child.pid, port: VITE_PORT, started: new Date().toISOString() } };
};

export const verifyStopDevLoop: QueryHandler = async (_args, projectDir) => {
  const pidPath = join(projectDir, '.dev-logs', 'dev-server.pid');
  if (!existsSync(pidPath)) return { data: { error: 'no dev server running' } };
  const pid = Number(readFileSync(pidPath, 'utf8').trim());
  if (!pid) return { data: { error: 'pid file empty or invalid' } };
  try { process.kill(-pid, 'SIGTERM'); } catch { /* group may already be dead */ }
  try { unlinkSync(pidPath); } catch {}
  try { unlinkSync(join(projectDir, '.dev-logs', 'dev-server.port')); } catch {}
  return { data: { stopped: pid } };
};
```
