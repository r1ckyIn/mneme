```rust
use std::io;
use std::marker::PhantomData;
use std::path::Path;

pub struct ImportToken {
    // CYCLE-2 cluster #5: field is PRIVATE (no `pub(crate)`). Only `import_handle()`
    // inside this module can construct an ImportToken; sibling modules under
    // src-tauri/src/ CANNOT write `ImportToken { _marker: PhantomData }` literal
    // (the field is invisible outside vault_writer.rs).
    _marker: PhantomData<()>,
}

pub enum WriteContext {
    User,
    Import(ImportToken),
}

#[derive(Debug, thiserror::Error)]
pub enum VaultWriterError {
    #[error("write to _source/ forbidden under User context")]
    WriteToSourceForbidden,
    #[error("path canonicalize failed: {0}")]
    CanonicalizeFailed(io::Error),
    #[error("permission failed: {0}")]
    PermissionFailed(io::Error),
    #[error("io failed: {0}")]
    IoFailed(io::Error),
    #[error("scaffold failed at {0:?}: {1}")]
    ScaffoldFailed(std::path::PathBuf, io::Error),
    #[error("invalid course code: {0}")]
    InvalidCourseCode(String),
    // CYCLE-2 cluster #8 — new structured variant for _source/ basename collisions.
    // import_controller.rs (02-05) uses this to surface a friendly "file already
    // exists" message via the import-error.ts classifier (02-11) instead of
    // silently overwriting the user's source files.
    #[error("source basename clash: a file named {0:?} already exists in _source/")]
    SourceBasenameClash(std::path::PathBuf),
}

pub fn import_handle() -> ImportToken;

pub fn write_to_vault(
    path: &Path,
    bytes: &[u8],
    ctx: WriteContext,
) -> Result<(), VaultWriterError>;

pub fn create_vault_scaffold(root: &Path) -> Result<(), VaultWriterError>;

pub fn create_course(root: &Path, code: &str) -> Result<(), VaultWriterError>;

pub fn with_temporary_writable_permission<F, R>(
    path: &Path,
    f: F,
) -> Result<R, VaultWriterError>
where
    F: FnOnce() -> Result<R, VaultWriterError>;
```

use std::fs;
use std::os::unix::fs::symlink;
use tempfile::tempdir;

#[test]
fn rejects_symlink_pointing_into_source() {
    let td = tempdir().expect("tempdir");
    let vault = td.path();
    fs::create_dir_all(vault.join("courses/COMP3221/_source")).unwrap();
    fs::create_dir_all(vault.join("courses/COMP3221/notes")).unwrap();
    let target = vault.join("courses/COMP3221/_source/secret.pdf");
    fs::write(&target, b"orig").unwrap();
    let link = vault.join("courses/COMP3221/notes/disguise.pdf");
    symlink(&target, &link).unwrap();
    let err = mneme_lib::vault_writer::write_to_vault(
        &link,
        b"new",
        mneme_lib::vault_writer::WriteContext::User,
    )
    .unwrap_err();
    assert!(
        matches!(err, mneme_lib::vault_writer::VaultWriterError::WriteToSourceForbidden),
        "expected WriteToSourceForbidden after canonicalize, got {:?}", err
    );
    // CYCLE-2 cluster #3 defense-in-depth: the _source/ target's bytes must NOT
    // have been overwritten through the symlink path, AND the symlink itself must
    // still exist (i.e. not replaced by a regular file).
    assert_eq!(
        fs::read(&target).unwrap(),
        b"orig",
        "_source/secret.pdf was overwritten through the symlink — guard bypassed!"
    );
    let link_meta = fs::symlink_metadata(&link).unwrap();
    assert!(
        link_meta.file_type().is_symlink(),
        "symlink leaf was replaced with a regular file — defense-in-depth invariant violated"
    );
}

// Maps to: SPEC REQ-03 (_source/ write protection — enum guard), REQ-04 (chmod 0o444),

//          D-07 chmod 644 → write → sync → 444 three-step.

//   std::fs::canonicalize requires the path to exist. The target file does NOT exist
//   at guard time (we are about to create it). Canonicalizing the parent dir +
//   re-attaching the basename resolves symlinks + .. components in the parent,
//   without requiring the final segment to exist. This catches the `..` traversal
//   that motivates T-2-01 AND the symlink-into-_source/ attack that motivates T-2-02.

// Why chmod 0o444 last, not first:

// ---------------------------------------------------------------------------
// Private token — only this module's `import_handle()` can construct.
// ---------------------------------------------------------------------------

pub struct ImportToken {
    // CYCLE-2 cluster #5 — field is fully PRIVATE (no `pub(crate)`). Sibling modules
    // under `src-tauri/src/` (config.rs, onboarding.rs, vault_index.rs,
    // import_controller.rs, lib.rs) CANNOT construct `ImportToken { _marker: ... }`
    // because the field is invisible outside this module. The factory `import_handle()`
    // is the only construction path. Audit grep gate at audit-capabilities.sh: literal
    // `ImportToken {` outside vault_writer.rs MUST be rejected.
    _marker: PhantomData<()>,
}

pub enum WriteContext {
    User,
    Import(ImportToken),
}

#[derive(Debug, thiserror::Error)]
pub enum VaultWriterError {
    #[error("write to _source/ forbidden under User context")]
    WriteToSourceForbidden,
    #[error("path canonicalize failed: {0}")]
    CanonicalizeFailed(std::io::Error),
    #[error("permission failed: {0}")]
    PermissionFailed(std::io::Error),
    #[error("io failed: {0}")]
    IoFailed(std::io::Error),
    #[error("scaffold failed at {0:?}: {1}")]
    ScaffoldFailed(PathBuf, std::io::Error),
    #[error("invalid course code: {0}")]
    InvalidCourseCode(String),
    // CYCLE-2 cluster #8 — structured variant for _source/ basename collisions.
    #[error("source basename clash: a file named {0:?} already exists in _source/")]
    SourceBasenameClash(PathBuf),
}

static VALID_COURSE_CODE: Lazy<Regex> =
    Lazy::new(|| Regex::new(r"^[A-Z]{4}\d{4}$").expect("compile course-code regex"));

// ---------------------------------------------------------------------------
// import_handle — the one factory the import controller calls.
// Audit gate: `grep -n 'import_handle()' src-tauri/src/` surfaces every privileged caller.
// ---------------------------------------------------------------------------

pub fn import_handle() -> ImportToken {
    // Construction is legal here because the literal lives in the same module
    // where ImportToken._marker is declared. Outside this module, the same syntax
    // is rejected by rustc with E0451 "field `_marker` of struct `ImportToken` is private".
    ImportToken { _marker: PhantomData }
}

// ---------------------------------------------------------------------------
// Path guard — canonicalize the PARENT directory + re-attach file_name.
// ---------------------------------------------------------------------------

fn is_under_source(path: &Path) -> Result<bool, VaultWriterError> {
    // CYCLE-2 cluster #3: two-arm canonicalization closes the leaf-symlink gap.
    // Arm A (path exists, possibly as a symlink): canonicalize the WHOLE path so
    // `symlink_metadata.is_symlink() == true` resolves through the link to the real
    // _source/ target. This catches T-2-02 (symlink in notes/ → _source/secret.pdf).
    // Arm B (path does not exist — new file create): canonicalize the PARENT then
    // re-attach file_name. This catches T-2-01 (`..` traversal during fs::create).
    let canon_full: PathBuf = if path.symlink_metadata().is_ok() {
        // Arm A — leaf exists (regular file, dir, or symlink). canonicalize follows
        // symlinks and normalizes `..` segments in one operation.
        path.canonicalize().map_err(VaultWriterError::CanonicalizeFailed)?
    } else {
        // Arm B — leaf does not exist; canonicalize parent + join basename.
        let parent = path.parent().ok_or_else(|| {
            VaultWriterError::CanonicalizeFailed(std::io::Error::new(
                std::io::ErrorKind::InvalidInput,
                "path has no parent",
            ))
        })?;
        let canon_parent = parent.canonicalize().map_err(VaultWriterError::CanonicalizeFailed)?;
        match path.file_name() {
            Some(name) => canon_parent.join(name),
            None => canon_parent,
        }
    };

    // Path-component check (NOT substring) — RESEARCH Pitfall: a path containing the
    // literal segment `_source` as a file (e.g. "/vault/notes/my_source") must NOT
    // match. The cycle-2 cluster #3 hardening: iterate Components, look for an exact
    // OsStr equal to "_source".
    for comp in canon_full.components() {
        if let std::path::Component::Normal(seg) = comp {
            if seg == std::ffi::OsStr::new("_source") {
                return Ok(true);
            }
        }
    }
    Ok(false)
}

// ---------------------------------------------------------------------------
// write_to_vault — the only public writer surface.
// ---------------------------------------------------------------------------

pub fn write_to_vault(
    path: &Path,
    bytes: &[u8],
    ctx: WriteContext,
) -> Result<(), VaultWriterError> {
    let under_source = is_under_source(path)?;
    match ctx {
        WriteContext::User => {
            if under_source {
                return Err(VaultWriterError::WriteToSourceForbidden);
            }
            atomic_write(path, bytes)?;
        }
        WriteContext::Import(_token) => {
            // Privileged — write then chmod 0o444 if under _source/.
            atomic_write(path, bytes)?;
            if under_source {
                let perms = std::fs::Permissions::from_mode(0o444);
                fs::set_permissions(path, perms).map_err(VaultWriterError::PermissionFailed)?;
            }
        }
    }
    Ok(())
}

// ---------------------------------------------------------------------------
// atomic_write — temp+rename helper (Phase 0 LEARNINGS rule).
// ---------------------------------------------------------------------------

fn atomic_write(path: &Path, bytes: &[u8]) -> Result<(), VaultWriterError> {
    let tmp = path.with_extension(format!(
        "{}.tmp",
        path.extension().and_then(|s| s.to_str()).unwrap_or("")
    ));
    {
        let mut f = fs::File::create(&tmp).map_err(VaultWriterError::IoFailed)?;
        f.write_all(bytes).map_err(VaultWriterError::IoFailed)?;
        f.sync_all().map_err(VaultWriterError::IoFailed)?;
    }
    fs::rename(&tmp, path).map_err(VaultWriterError::IoFailed)?;
    Ok(())
}

// ---------------------------------------------------------------------------
// Vault scaffold + course scaffold (idempotent).
// ---------------------------------------------------------------------------

pub fn create_vault_scaffold(root: &Path) -> Result<(), VaultWriterError> {
    for sub in &["_system", "_inbox", "courses", "shared"] {
        let p = root.join(sub);
        fs::create_dir_all(&p)
            .map_err(|e| VaultWriterError::ScaffoldFailed(p.clone(), e))?;
    }
    Ok(())
}

pub fn create_course(root: &Path, code: &str) -> Result<(), VaultWriterError> {
    if !VALID_COURSE_CODE.is_match(code) {
        return Err(VaultWriterError::InvalidCourseCode(code.to_string()));
    }
    let course_root = root.join("courses").join(code);
    for sub in &[
        "_source",
        "_source/lectures",
        "_source/tutorials",
        "_source/assignments",
        "notes",
        "concepts",
        "practice",
    ] {
        let p = course_root.join(sub);
        fs::create_dir_all(&p)
            .map_err(|e| VaultWriterError::ScaffoldFailed(p.clone(), e))?;
    }
    let index_md = course_root.join("INDEX.md");
    if !index_md.exists() {
        let created = Utc::now().to_rfc3339();
        let body = format!("---\ncourse: {}\ncreated: {}\n---\n", code, created);
        atomic_write(&index_md, body.as_bytes())?;
    }
    Ok(())
}

// with_temporary_writable_permission — Phase 2 D-07.
// chmod 0o644 → run inner → chmod 0o444 (via Drop on guard).
// Drop guard fires even on panic / early-return / cancellation (Pitfall 12).
// ---------------------------------------------------------------------------

struct RelockGuard<'a> {
    path: &'a Path,
    // If `armed = true`, Drop sets perms back to 0o444. The wrapped operation can
    // call `guard.disarm()` after the final 0o444 chmod has been applied normally,
    // so Drop becomes a no-op on the happy path. On panic / early return without
    // disarm, Drop re-applies 0o444 as a defense-in-depth relock.
    armed: bool,
}

impl<'a> RelockGuard<'a> {
    fn new(path: &'a Path) -> Self {
        Self { path, armed: true }
    }
    fn disarm(&mut self) {
        self.armed = false;
    }
}

impl<'a> Drop for RelockGuard<'a> {
    fn drop(&mut self) {
        if self.armed {
            // Best-effort relock — if this fails the file stays at 0o644 + the original
            // Err path informs the caller (D-07 contract). Log via eprintln per Phase 1
            // convention (no logging crate).
            let _ = fs::set_permissions(self.path, fs::Permissions::from_mode(0o444));
        }
    }
}

pub fn with_temporary_writable_permission<F, R>(
    path: &Path,
    f: F,
) -> Result<R, VaultWriterError>
where
    F: FnOnce() -> Result<R, VaultWriterError>,
{
    // Step 1 — chmod 0o644 (unlock).
    fs::set_permissions(path, fs::Permissions::from_mode(0o644))
        .map_err(VaultWriterError::PermissionFailed)?;
    // Arm the Drop guard so any panic / early return relocks.
    let mut guard = RelockGuard::new(path);
    // Step 2 — caller-supplied write operation.
    let r = f()?;
    // Step 3 — chmod 0o444 (relock).
    fs::set_permissions(path, fs::Permissions::from_mode(0o444))
        .map_err(VaultWriterError::PermissionFailed)?;
    // Disarm guard — Drop becomes no-op on happy path.
    guard.disarm();
    Ok(r)
}
