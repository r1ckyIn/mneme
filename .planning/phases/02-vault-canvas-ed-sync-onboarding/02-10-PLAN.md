SettingsPanel.svelte structure (UI-SPEC §8.2):

- 880 × 600 px modal, centered, `<div role="dialog" aria-modal="true">` NOT `<dialog>` (UI-SPEC §8.2 chose div-overlay because 8-cat layout exceeds modal-native ergonomics).

- localStorage key `mneme.settings.activeCategory` persists last-active category across sessions.
- **CYCLE-3 priority #3 — ownership contract LOCKED to prop-controlled**: SettingsPanel is a PURE prop-controlled component. Props: `open: boolean` (parent owns the boolean) + `onClose: () => void` (parent flips it back to false on user close). Cmd+, is implemented as a `<svelte:window onkeydown={...}>` handler INSIDE SettingsPanel that dispatches `mneme:open-settings` as a window CustomEvent. The PARENT (02-12 +page.svelte) listens for `mneme:open-settings` AND `menu:open-settings` (Tauri event → CustomEvent bridge in +layout.svelte per SPEC-GAP-1) AND wires both to `settingsOpen = true`. SettingsPanel does NOT subscribe to `mneme:open-settings` itself; it only DISPATCHES. The cycle-2 "SettingsPanel owns state via its own listener" contract is REJECTED — it duplicated state machines with the parent and broke the prop interface.
- NO `installSettingsShortcut()` export. The Cmd+, key handler is a single `<svelte:window>` directive scoped to SettingsPanel — installed once when the component mounts, cleaned up automatically on unmount. The handler runs even when the panel is closed (svelte:window is window-global per component lifetime).
- Esc + backdrop-click close the panel.

**Step 4 — Create `src/lib/components/settings/VaultCategory.svelte`** (full feature — path display + Move + course list):

```svelte
<!--
  Visual: /Users/qinyuan/Downloads/Mneme 3/Mneme Settings.html (Vault body)
  + 02-UI-SPEC.md §8.2.1
  Vault path + Move (inline confirmation) + courses list + Add course inline form.
  Reads from vault-state.svelte.ts singleton; writes via invoke().
-->
<script lang="ts">
  import { onMount } from "svelte";
  import { invoke } from "@tauri-apps/api/core";
  import { open } from "@tauri-apps/plugin-dialog";
  import { getVaultState, setVaultPath, setCourseList, addCourse, removeCourse } from "$lib/vault-state.svelte";
  import { validateCourseCode } from "$lib/onboarding-validation";

  const vaultState = getVaultState();
  let moveConfirming = $state<{ newPath: string } | null>(null);
  let moveStatus = $state<"idle" | "moving" | "done" | "error">("idle");
  let moveError = $state<string | null>(null);
  let addingCourse = $state(false);
  let courseInput = $state("");
  let lastError = $state<string | null>(null);

  onMount(async () => {
    try {
      const cfg = await invoke<{ vault_path: string }>("load_config");
      if (cfg.vault_path) setVaultPath(cfg.vault_path);
      const courses = await invoke<string[]>("list_courses");
      setCourseList(courses);
    } catch (e) {
      console.error("[settings:vault-load]", e);
    }
  });

  async function browseAndMove() {
    try {
      const chosen = await open({ directory: true, multiple: false });
      if (typeof chosen === "string") {
        moveConfirming = { newPath: chosen };
      }
    } catch (e) {
      console.error("[settings:browse-move]", e);
    }
  }

  async function confirmMove() {
    if (!moveConfirming) return;
    moveStatus = "moving";
    moveError = null;
    try {
      const summary = await invoke<{ files_copied: number; bytes_copied: number; old_root_preserved: boolean }>(
        "move_vault",
        { oldRoot: vaultState.vault_path, newRoot: moveConfirming.newPath },
      );
      setVaultPath(moveConfirming.newPath);
      await invoke("save_config", { state: { vault_path: moveConfirming.newPath, schema_version: 1 } });
      moveStatus = "done";
      moveConfirming = null;
      console.log("[settings:move-vault] complete", summary);
    } catch (e) {
      moveStatus = "error";
      moveError = String(e);
      console.error("[settings:move-vault]", e);
    }
  }

  function cancelMove() {
    moveConfirming = null;
    moveStatus = "idle";
  }

  async function tryAddCourse() {
    if (validateCourseCode(courseInput).kind !== "valid") return;
    const code = courseInput.trim();
    try {
      await invoke("course_create", { root: vaultState.vault_path, code });
      addCourse(code);
      courseInput = "";
      lastError = null;
    } catch (e) {
      lastError = `Couldn't create that course folder. ${String(e)}`;
    }
  }

  async function tryRemoveCourse(code: string) {
    // Phase 2 SPEC: removing from settings does NOT delete files on disk; UI-only.
    // Future Phase 3 may wire a real `course_remove` IPC; for v1 we just drop from list.
    removeCourse(code);
  }
</script>

<div class="vault">
  <h2 class="cat-heading">Vault</h2>

    <div class="group-label">Vault path</div>
    <input type="text" class="path-input" value={vaultState.vault_path || "Not set"} readonly aria-label="Vault path" />
    <div class="row-actions">
      <button type="button" class="ghost-btn" onclick={browseAndMove}>Browse…</button>
      <button type="button" class="ghost-btn" onclick={browseAndMove}>Move…</button>
    </div>
    {#if moveStatus === "done"}
      <p class="status success">Move complete. The original folder stays at the previous location for you to delete manually in Finder.</p>
    {:else if moveStatus === "error" && moveError}
      <p class="status error">{moveError}</p>
    {/if}
  </section>

  {#if moveConfirming}
    <div class="confirm-overlay">
      <div class="confirm-card">
        <p class="confirm-text">
          Move vault to <code>{moveConfirming.newPath}</code>?<br />
          The original folder stays at <code>{vaultState.vault_path}</code> for you to delete manually in Finder.
        </p>
        <div class="confirm-actions">
          <button type="button" class="ghost-btn" onclick={cancelMove}>Cancel</button>
          <button type="button" class="cta" onclick={confirmMove} disabled={moveStatus === "moving"}>Move vault</button>
        </div>
      </div>
    </div>
  {/if}

  <div class="section-divider"></div>

  <section class="group">
    <div class="group-label">Courses</div>
    <div class="course-list">
      {#each vaultState.course_list as code (code)}
        <div class="course-row">
          <span class="code">{code}</span>
          <span class="note-count">notes (—)</span>
          <button type="button" class="row-x" onclick={() => tryRemoveCourse(code)} aria-label={`Remove ${code}`}>⋯</button>
        </div>
      {/each}
    </div>

    {#if addingCourse}
      <div class="add-form">
        <input
          type="text"
```
