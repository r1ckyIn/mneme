```svelte
<script lang="ts">
  import { onMount } from "svelte";
  import { goto } from "$app/navigation";
  import { invoke } from "@tauri-apps/api/core";
  import Step1Welcome from "./Step1Welcome.svelte";
  // Step2-3-5-6 lazy-imported in Plan 09; this plan uses a Placeholder stub.
  import Step4MCPStatus from "./Step4MCPStatus.svelte";
  import OnboardingStepRail from "./OnboardingStepRail.svelte";

  interface Props {
    initialStep: number;
  }
  let { initialStep }: Props = $props();

  interface OnboardingState {
    current_step: number;
    vault_path: string;
    courses_added: string[];
    completed_at: string | null;
  }

  let onboardingState = $state<OnboardingState>({
    current_step: initialStep,
    vault_path: "",
    courses_added: [],
    completed_at: null,
  });
  let saving = $state(false);

  onMount(async () => {
    try {
      const loaded = await invoke<OnboardingState>("load_onboarding_state");
      onboardingState = loaded;
    } catch (e) {
      console.error("[onboarding:load]", e);
    }
  });

  async function next() {
    if (saving) return;
    saving = true;
    try {
      // Move forward; clamp at 6.
      const nextStep = Math.min(6, onboardingState.current_step + 1);
      onboardingState.current_step = nextStep;
      await invoke("save_onboarding_state", { state: onboardingState });
      await goto(`/onboarding/${nextStep}`, { replaceState: true });
    } catch (e) {
      console.error("[onboarding:next]", e);
    } finally {
      saving = false;
    }
  }

  async function finish() {
    if (saving) return;
    saving = true;
    try {
      await invoke("complete_onboarding");
      await goto("/", { replaceState: true });
    } catch (e) {
      console.error("[onboarding:finish]", e);
    } finally {
      saving = false;
    }
  }
</script>
```

**CYCLE-2 cluster #11 FIX**: cycle-1 codex review correctly flagged that `ready = false` → `await goto(...)` → `return` permanently blocks render because SvelteKit root layouts do NOT remount on same-root navigation; `+layout.svelte` is mounted ONCE for the whole app lifecycle. The original snippet would blank-screen fresh installs forever.

The fix has two prongs:

(a) **Set `ready = true` BEFORE the `goto`** so the onboarding child route renders immediately. The onboarding route has its own `/onboarding/+layout.svelte` (no MainUI splitter/MindMapBar), so showing children early is correct — the main UI shell is NOT what we'd want to render under `/onboarding/*` anyway.

(b) **Watch the page store via `$effect`** so navigating BACK from onboarding to `/` (after `complete_onboarding`) doesn't try to re-load state in an inconsistent way. This is forward-compat for Plan 09 Step 6 Finish flow.

Final snippet:

```svelte
<script lang="ts">
  import "$lib/styles/tokens.css";
  import "katex/dist/katex.min.css";
  import { onMount } from "svelte";
  import { invoke } from "@tauri-apps/api/core";
  import { goto } from "$app/navigation";
  import { page } from "$app/stores";

  if (import.meta.env.DEV) {
    void import("$lib/dev/console-forwarder").then(({ installConsoleForwarder }) => {
      installConsoleForwarder();
    });
  }

  let { children } = $props();
  let ready = $state(false);

  onMount(async () => {
    try {
      const state = await invoke<{ current_step: number; completed_at: string | null }>(
        "load_onboarding_state"
      );
      const onCurrentRoute = $page.url.pathname.startsWith("/onboarding");
      if (!state.completed_at && !onCurrentRoute) {
        // CYCLE-2 cluster #11 — set ready BEFORE goto so the onboarding child
        // route renders immediately on the SAME mounted root layout instance.
        // SvelteKit root layouts do NOT remount on same-root navigation, so
        // leaving ready=false here would blank-screen first-launch users forever.
        ready = true;
        await goto(`/onboarding/${state.current_step || 1}`, { replaceState: true });
        return;
      }
    } catch (e) {
      console.error("[+layout:load_onboarding]", e);
    }
    ready = true;
  });
</script>

{#if ready}
  {@render children()}
{/if}
```

(Phase 1's `+layout.svelte` does NOT block on this; if Phase 1 already renders eagerly, wrap the existing render in `{#if ready}` and gate.)

**Why not SvelteKit `+page.ts load()` redirect** (the alternative cycle-1 suggested): the root `+layout.svelte` installs the DEV console forwarder, which MUST run before any onboarding child fetches anything. Moving the redirect to `load()` would put the DEV forwarder install inside a `load()` function (deprecated pattern — `load` is for data, not side-effects). The `ready` flag fix is mechanically smaller and preserves the proven DEV-forwarder install order from Phase 01.1. Decision: option-A `ready=true-then-goto`.
