// src/lib/import-state-derive.ts — pure derivation, no Svelte runes, no IPC.
// Maps to: SPEC REQ-13 (status pill states); UI-SPEC §8.5 pill copy contract.
// Why a separate file: keeps Vitest tests jsdom-mount-free. Tests run on Node fast path.

export interface ImportProgress {
    operation_id: string;
    current: number;
    total: number;
    last_file_name: string;
    last_file_status: "ok" | "error" | "cancelled" | "done";
    last_file_error: string | null;
}

// CYCLE-2 cluster #6 — SSOT mirror.
// Source: src-tauri/src/import_controller.rs ImportDoneEvent (02-05 PLAN).
// Field names MUST match Rust verbatim (snake_case): operation_id, total,
// succeeded, failed, cancelled, course, category, failures.
// `failures[].reason` (NOT .message) — cycle-2 cluster #6 field rename.
// When Rust ImportDoneEvent changes, this type changes in the SAME PR.
export interface ImportFailure {
    path: string;
    reason: string;          // CYCLE-2 cluster #6 — renamed from `message` to match Rust ImportFailure.reason
}

export interface ImportHistoryEntry {
    operation_id: string;
    timestamp_iso: string;   // Frontend-derived (Tauri ImportDoneEvent does NOT carry timestamp; listener adds at receipt time)
    total: number;
    succeeded: number;
    failed: number;
    cancelled: boolean;
    course: string | null;
    category: string;
    failures: ImportFailure[];   // CYCLE-2 cluster #6 — ALWAYS array (never undefined); Rust emits empty Vec on success
}

export type PillState =
    | { kind: "idle" }
    | { kind: "importing"; current: number; total: number }
    | { kind: "imported"; count: number; recency: string }
    | { kind: "partial"; succeeded: number; failed: number }
    | { kind: "cancelled"; succeeded: number; total: number };

const STALE_HISTORY_MS = 24 * 3600 * 1000;  // 1 day — after this, pill returns to idle

export function derivePillState(
    progress: ImportProgress | null,
    recent: ImportHistoryEntry[],
    now: Date,
): PillState {
    // Active import wins.
    if (progress && progress.last_file_status !== "done") {
        return { kind: "importing", current: progress.current, total: progress.total };
    }
    const latest = recent[0];
    if (!latest) return { kind: "idle" };

    const age = now.getTime() - new Date(latest.timestamp_iso).getTime();
    if (age >= STALE_HISTORY_MS) return { kind: "idle" };

    if (latest.cancelled) {
        return { kind: "cancelled", succeeded: latest.succeeded, total: latest.total };
    }
    if (latest.failed > 0) {
        return { kind: "partial", succeeded: latest.succeeded, failed: latest.failed };
    }
    return { kind: "imported", count: latest.succeeded, recency: formatRecency(latest.timestamp_iso, now) };
}

export function formatRecency(timestampIso: string, now: Date): string {
    const age_s = Math.floor((now.getTime() - new Date(timestampIso).getTime()) / 1000);
    if (age_s < 60) return "just now";
    const age_m = Math.floor(age_s / 60);
    if (age_m < 60) return `${age_m}m ago`;
    const age_h = Math.floor(age_m / 60);
    if (age_h < 24) return `${age_h}h ago`;
    const age_d = Math.floor(age_h / 24);
    return `${age_d}d ago`;
}
