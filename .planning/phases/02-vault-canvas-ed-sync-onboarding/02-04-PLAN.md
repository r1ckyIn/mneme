// src-tauri/src/vault_index.rs — Phase 2 D-14 + REQ-13 vault file index.
// Maps to: SPEC REQ-13 (~/.mneme/vault-index.db; filesystem is SSOT, index is optimization).
// Pattern: RESEARCH §Pattern 4. SQL injection mitigation: params![] on every query.
// Threading: std::sync::Mutex<Connection> (no .await crosses lock — Pitfall 11).

use std::fs;
use std::path::{Path, PathBuf};
use std::sync::Mutex;

use chrono::Utc;
use rusqlite::{params, Connection};

pub struct VaultIndex {
    conn: Mutex<Connection>,
}

#[derive(Debug, thiserror::Error)]
pub enum VaultIndexError {
    #[error("sqlite: {0}")]
    Sqlite(#[from] rusqlite::Error),
    #[error("io: {0}")]
    Io(#[from] std::io::Error),
    #[error("walkdir: {0}")]
    Walk(#[from] walkdir::Error),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct VaultFileRow {
    pub path: String,
    pub course: Option<String>,
    pub kind: String,
    pub size_bytes: i64,
    pub mtime_iso: String,
    pub indexed_at_iso: String,
}

#[derive(Debug, Clone, Default)]
pub struct ReconcileSummary {
    pub inserted: usize,
    pub deleted: usize,
    pub scanned: usize,
}

impl VaultIndex {
    pub fn init(db_path: &Path) -> Result<Self, VaultIndexError> {
        if let Some(parent) = db_path.parent() { fs::create_dir_all(parent)?; }
        let conn = Connection::open(db_path)?;
        // Pragma order matters (RESEARCH L583-586).
        conn.pragma_update(None, "journal_mode", "WAL")?;
        conn.pragma_update(None, "synchronous", "NORMAL")?;
        conn.pragma_update(None, "busy_timeout", 5000_i32)?;
        conn.pragma_update(None, "wal_autocheckpoint", 1000_i32)?;
        conn.execute_batch(
            r#"
            CREATE TABLE IF NOT EXISTS vault_files (
                path TEXT PRIMARY KEY,
                course TEXT,
                kind TEXT,
                size_bytes INTEGER,
                mtime_iso TEXT,
                indexed_at_iso TEXT
            );
            CREATE INDEX IF NOT EXISTS vault_files_course_idx ON vault_files(course);
            "#,
        )?;
        Ok(VaultIndex { conn: Mutex::new(conn) })
    }

    fn with_conn<F, R>(&self, f: F) -> Result<R, VaultIndexError>
    where F: FnOnce(&Connection) -> Result<R, VaultIndexError>,
    {
        // Phase 1 WR-01 poisoned-mutex tolerance.
        let guard = self.conn.lock().unwrap_or_else(|p| {
            eprintln!("[vault_index] mutex poisoned — recovering inner state");
            p.into_inner()
        });
        f(&guard)
    }

    pub fn insert(&self, row: &VaultFileRow) -> Result<(), VaultIndexError> {
        self.with_conn(|c| {
            c.execute(
                r#"INSERT OR REPLACE INTO vault_files
                   (path, course, kind, size_bytes, mtime_iso, indexed_at_iso)
                   VALUES (?1, ?2, ?3, ?4, ?5, ?6)"#,
                params![row.path, row.course, row.kind, row.size_bytes, row.mtime_iso, row.indexed_at_iso],
            )?;
            Ok(())
        })
    }

    pub fn get(&self, path: &str) -> Result<Option<VaultFileRow>, VaultIndexError> {
        self.with_conn(|c| {
            let mut stmt = c.prepare(
                r#"SELECT path, course, kind, size_bytes, mtime_iso, indexed_at_iso
                   FROM vault_files WHERE path = ?1"#,
            )?;
            let mut rows = stmt.query(params![path])?;
            if let Some(r) = rows.next()? {
                Ok(Some(VaultFileRow {
                    path: r.get(0)?, course: r.get(1)?, kind: r.get(2)?,
                    size_bytes: r.get(3)?, mtime_iso: r.get(4)?, indexed_at_iso: r.get(5)?,
                }))
            } else { Ok(None) }
        })
    }

    pub fn delete(&self, path: &str) -> Result<(), VaultIndexError> {
        self.with_conn(|c| {
            c.execute("DELETE FROM vault_files WHERE path = ?1", params![path])?;
            Ok(())
        })
    }

    pub fn count_all(&self) -> Result<i64, VaultIndexError> {
        self.with_conn(|c| {
            let n: i64 = c.query_row("SELECT COUNT(*) FROM vault_files", [], |row| row.get(0))?;
            Ok(n)
        })
    }

    pub fn list_courses(&self) -> Result<Vec<String>, VaultIndexError> {
        self.with_conn(|c| {
            let mut stmt = c.prepare(
                "SELECT DISTINCT course FROM vault_files WHERE course IS NOT NULL ORDER BY course ASC",
            )?;
            let rows = stmt.query_map([], |row| row.get::<_, String>(0))?;
            let mut out = Vec::new();
            for r in rows { out.push(r?); }
            Ok(out)
        })
    }

    pub fn list_all_paths(&self) -> Result<Vec<String>, VaultIndexError> {
        self.with_conn(|c| {
            let mut stmt = c.prepare("SELECT path FROM vault_files")?;
            let rows = stmt.query_map([], |row| row.get::<_, String>(0))?;
            let mut out = Vec::new();
            for r in rows { out.push(r?); }
            Ok(out)
        })
    }

    // CYCLE-3 priority #9 — REMOVED `locked_conn` helper. Integration tests under
    // `src-tauri/tests/` cannot reach `pub(crate)`. The vault_index_count.rs test opens
    // a SECOND `rusqlite::Connection::open(&db_path)` instead — SQLite WAL mode
    // permits multi-connection reads on the same DB file. This keeps the internal
    // Mutex<Connection> fully private.

    // reconcile defined in Task 2 (same `impl VaultIndex` block).
}

pub fn classify_kind(path: &Path, vault_root: &Path) -> String {
    let rel = path.strip_prefix(vault_root).unwrap_or(path);
    let s = rel.to_string_lossy();
    if s.contains("/_source/") || s.ends_with("/_source") { return "_source".into(); }
    if s.contains("/notes/") || s.ends_with("/notes") { return "notes".into(); }
    if s.contains("/concepts/") || s.ends_with("/concepts") { return "concepts".into(); }
    if s.contains("/practice/") || s.ends_with("/practice") { return "practice".into(); }
    if s.ends_with("INDEX.md") { return "INDEX".into(); }
    if s.starts_with("_inbox") { return "_inbox".into(); }
    if s.starts_with("_system") { return "_system".into(); }
    if s.starts_with("shared") { return "shared".into(); }
    "other".into()
}

pub fn extract_course(path: &Path, vault_root: &Path) -> Option<String> {
    let rel = path.strip_prefix(vault_root).ok()?;
    let mut components = rel.components();
    let first = components.next()?;
    if first.as_os_str() != "courses" { return None; }
    Some(components.next()?.as_os_str().to_string_lossy().to_string())
}

pub fn now_iso() -> String { Utc::now().to_rfc3339() }

// Append inside `impl VaultIndex`:

impl VaultIndex {
    // ... existing methods from Task 1 ...

    pub fn reconcile(&self, vault_root: &Path) -> Result<ReconcileSummary, VaultIndexError> {
        use std::collections::HashSet;
        let mut summary = ReconcileSummary::default();
        let now = now_iso();

        // 1) Walk the on-disk vault, collect existing files + insert new rows.
        // CYCLE-3 priority #9 — filter_entry prunes hidden DIRECTORIES before descent
        // (cycle-2 H2: the cycle-2 version only filtered hidden FILES, so .mneme/
        // and .git/ subtrees got walked + their internal files indexed). The
        // load-bearing names to prune: `.mneme` (vault-index.db + WAL), `.git`,
        // `.DS_Store`, any other `.`-prefixed dir. We keep the per-file `.`-prefix
        // skip too as defense-in-depth for files at vault root.
        fn is_hidden_entry(e: &walkdir::DirEntry) -> bool {
            e.file_name()
                .to_str()
                .map_or(false, |s| s.starts_with('.'))
        }
        let mut on_disk: HashSet<String> = HashSet::new();
        let walker = walkdir::WalkDir::new(vault_root)
            .into_iter()
            .filter_entry(|e| !is_hidden_entry(e));
        for entry in walker.filter_map(Result::ok) {
            if !entry.file_type().is_file() { continue; }
            let p = entry.path();
            // Defense-in-depth: skip hidden files (e.g. orphans created outside
            // a .git/ subdir we already pruned).
            if p.file_name().map_or(false, |n| n.to_string_lossy().starts_with('.')) { continue; }
            summary.scanned += 1;
            let canonical = match p.canonicalize() { Ok(c) => c, Err(_) => continue };
            let path_str = canonical.to_string_lossy().to_string();
            on_disk.insert(path_str.clone());

            // If already indexed, skip (cheap path); otherwise insert.
            if self.get(&path_str)?.is_some() { continue; }
            let meta = match fs::metadata(&canonical) { Ok(m) => m, Err(_) => continue };
            let mtime_iso = meta.modified()
                .ok()
                .and_then(|t| {
                    chrono::DateTime::<chrono::Utc>::from(t).to_rfc3339().into()
                })
                .unwrap_or_else(|| now.clone());
            let kind = classify_kind(&canonical, vault_root);
            let course = extract_course(&canonical, vault_root);
            self.insert(&VaultFileRow {
                path: path_str,
                course,
                kind,
                size_bytes: meta.len() as i64,
                mtime_iso,
                indexed_at_iso: now.clone(),
            })?;
            summary.inserted += 1;
        }

        // 2) Lazy-delete: any indexed path NOT on disk → drop.
        let all = self.list_all_paths()?;
        for p in all {
            if !on_disk.contains(&p) {
                self.delete(&p)?;
                summary.deleted += 1;
            }
        }

        Ok(summary)
    }
}

pub fn enumerate_folder_one_level(
    folder: &Path,
) -> Result<(Vec<PathBuf>, Vec<PathBuf>), VaultIndexError> {
    let mut files = Vec::new();
    let mut skipped_subdirs = Vec::new();
    for entry in fs::read_dir(folder)? {
        let entry = entry?;
        let name = entry.file_name();
        let name_str = name.to_string_lossy();
        // Hidden file skip (Pitfall: `.DS_Store` would pollute the import list).
        if name_str.starts_with('.') { continue; }
        let ftype = entry.file_type()?;
        if ftype.is_file() {
            files.push(entry.path());
        } else if ftype.is_dir() {
            skipped_subdirs.push(entry.path());
        }
        // Symlinks etc. silently skipped.
    }
    Ok((files, skipped_subdirs))
}
