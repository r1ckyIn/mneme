**`src/lib/connection-state.ts` — A-10 shared rune (NEW):**

```typescript
// src/lib/connection-state.ts — A-10 shared connection state for the titlebar.
//
// Phase 1: starts as "disconnected", flips to "connecting" when ChatPanel
// (01-06) submits the first prompt, flips to "connected" on first stream_event
// or first text_delta, flips back to "disconnected" on subprocess close
// without a result event.
//
// Plan 01-06's ChatPanel imports this rune and updates .status at lifecycle
// transitions. TitlebarMeta reads .status and renders the dot color + label.

import { writable } from "svelte/store";

export type ConnectionStatus = "connected" | "connecting" | "disconnected";

// Svelte 5 runes don't compose across modules cleanly; use a writable store as
// the lingua-franca. ChatPanel + TitlebarMeta both `$store.status` for reactive
// reads. (Alternative: a Svelte 5 `$state` exported from a `.svelte.ts` module —
// but the writable-store form keeps the dep injection explicit.)

export const connectionState = $state<{ status: ConnectionStatus }>({
  status: "disconnected",
});

export function setStatus(status: ConnectionStatus) {
  connectionState.status = status;
}
```

Note: Svelte 5 `$state` exported from a non-component module requires a `.svelte.ts` filename. If that's incompatible with the executor's environment, fall back to the writable-store pattern shown above (the ConnectionState type stays the same; consumers do `import { connectionState } from "$lib/connection-state"` and read `connectionState.status`).

  ~120 LOC fresh write per RESEARCH §Wave 1 step 1; rejected svelte-splitpanes

  CSS Grid minmax + ratio clamp enforces UI-SPEC §"Pane proportions + dividers"
  min-width 200px per pane (translates to 0.20 ratio floor at 1024px window;
  acceptable approximation — exact px-floor would require ResizeObserver, deferred).

  localStorage key `mneme.layout.split` per SPEC L37 + CONTEXT.md D-02.
  Stored value: JSON {"leftRatio": number, "middleRatio": number}.

<script lang="ts">
  import { onMount } from "svelte";

  type Snippet = (() => any) | undefined;

  interface Props {
    left?: Snippet;
    middleTop?: Snippet;
    middleBottom?: Snippet;
    right?: Snippet;
    bottom?: Snippet;
  }

  let { left, middleTop, middleBottom, right, bottom }: Props = $props();

  const STORAGE_KEY = "mneme.layout.split";
  const HANDLE_WIDTH_PX = 4;
  const RATIO_MIN = 0.20;     // ~200px floor at 1024px window
  const RATIO_MAX = 0.50;     // never let a single pane eat more than half

  let leftRatio = $state(0.30);
  let middleRatio = $state(0.40);
  let dragging = $state<"left" | "right" | null>(null);

  // Restore on mount — try/catch swallows JSON.parse errors and falls back
  // to the 30/40/30 defaults declared above.
  onMount(() => {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (raw) {
        const parsed = JSON.parse(raw);
        if (typeof parsed.leftRatio === "number" && typeof parsed.middleRatio === "number") {
          leftRatio = clamp(parsed.leftRatio, RATIO_MIN, RATIO_MAX);
          middleRatio = clamp(parsed.middleRatio, RATIO_MIN, RATIO_MAX);
        }
      }
    } catch {
      // Default values stand — no surfacing, this is a best-effort restore.
    }
  });

  function clamp(x: number, lo: number, hi: number): number {
    return Math.max(lo, Math.min(hi, x));
  }

  function startDrag(which: "left" | "right", e: PointerEvent) {
    dragging = which;
    (e.currentTarget as HTMLElement).setPointerCapture(e.pointerId);
    e.preventDefault();
  }

  function onPointerMove(e: PointerEvent) {
    if (!dragging) return;
    const totalWidth = window.innerWidth;
    if (totalWidth <= 0) return;
    const xRatio = e.clientX / totalWidth;
    if (dragging === "left") {
      const newLeft = clamp(xRatio, RATIO_MIN, 1 - RATIO_MIN - RATIO_MIN);
      if (newLeft + middleRatio + RATIO_MIN <= 1) {
        leftRatio = newLeft;
      } else {
        leftRatio = newLeft;
        middleRatio = clamp(1 - newLeft - RATIO_MIN, RATIO_MIN, RATIO_MAX);
      }
    } else if (dragging === "right") {
      const newMiddle = clamp(xRatio - leftRatio, RATIO_MIN, RATIO_MAX);
      middleRatio = newMiddle;
    }
  }

  function endDrag(e: PointerEvent) {
    if (!dragging) return;
    try {
      localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify({ leftRatio, middleRatio })
      );
    } catch {
      // localStorage unavailable — best-effort persist.
    }
    (e.currentTarget as HTMLElement).releasePointerCapture(e.pointerId);
    dragging = null;
  }

  let rightRatio = $derived(Math.max(RATIO_MIN, 1 - leftRatio - middleRatio));
</script>

// src/lib/connection-state.ts — A-10 shared connection state for the titlebar.
//
// Phase 1 lifecycle:
//   - "disconnected" — initial state on app launch (no subprocess yet, or after exit)
//   - "connecting"   — flipped by ChatPanel (01-06) when sendPrompt() is called and
//                      the spawn is in flight; cleared on first stream_event/text_delta
//   - "connected"    — flipped on first stream_event/text_delta; held until
//                      cmd.on("close") fires
//
// TitlebarMeta (Task 7) reads .status to render the dot color + label.
// ChatPanel (01-06) imports setStatus() and calls it at lifecycle transitions.

export type ConnectionStatus = "connected" | "connecting" | "disconnected";

// Exported as a plain mutable object so consumers can wrap in Svelte 5 $state()
// at component-mount time:
//
//   <script lang="ts">
//     import { connectionState } from "$lib/connection-state";
//     // Component uses connectionState.status reactively via Svelte 5 fine-grained tracking
//   </script>
//
// Setter is the canonical write path — consumers MUST call setStatus() rather
// than mutating .status directly so future refactors (writable store / signal /
// custom rune) have a single edit point.

export const connectionState: { status: ConnectionStatus } = {
  status: "disconnected",
};

export function setStatus(status: ConnectionStatus): void {
  connectionState.status = status;
}

//   export const connectionState = writable<{ status: ConnectionStatus }>({ status: "disconnected" });
//   export const setStatus = (s: ConnectionStatus) => connectionState.set({ status: s });

// TitlebarMeta would then read `$connectionState.status`. Document the choice
