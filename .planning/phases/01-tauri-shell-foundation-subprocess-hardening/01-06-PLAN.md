```
async function sendPrompt() {
  if (!prompt.trim() || dispatch.isStreaming) return;

  // 1. Push user msg into state
  const userText = prompt.trim();
  dispatch.messages.push({
    id: uid(),
    role: "user",
    text: userText,
    streaming: false,
  });
  prompt = "";
  dispatch.isStreaming = true;
  dispatch.resultReceived = false;
  pulseDotVisible = true;

  // 2. Build the spawn — capability layer enforces validators
  const cmd = Command.create("claude-bin", buildClaudeArgs(userText));
  cmdRef = cmd;

  // 3. ALL listeners BEFORE spawn (spike landmine #5)
  let buffer = "";
  let firstTextDeltaSeen = false;

  cmd.stdout.on("data", (chunk: string) => {
    buffer += chunk;
    const parts = buffer.split("\n");
    buffer = parts.pop() ?? "";
    for (const raw of parts) {
      if (!raw.trim()) continue;
      try {
        const evt = JSON.parse(raw);
        // Streaming dot UX — flip off on first text_delta
        if (
          !firstTextDeltaSeen &&
          evt?.type === "stream_event" &&
          evt?.event?.delta?.type === "text_delta"
        ) {
          firstTextDeltaSeen = true;
          pulseDotVisible = false;
        }
        // TOKENICODE control_request interception — bypass-mode hang prevention
        if (evt?.type === "control_request") {
          // Phase 1 doesn't engage control_request; log + skip the dispatch arm
          console.log("[claude:control_request] skipped (Phase 1 baseline)", evt);
          continue;
        }
        dispatchEvent(evt, dispatch);
      } catch {
        // Malformed line — drop, never throw mid-stream (spike landmine #2)
      }
    }
  });

  cmd.stderr.on("data", (data: string) => {
    console.warn("[claude:stderr]", data);
  });

  cmd.on("error", (err) => {
    console.error("[claude:spawn-error]", err);
    // Surface a system error bubble; user sees the raw text per SPEC L93
    dispatch.messages.push({
      id: uid(),
      role: "system",
      text: escapeHtml(String(err)),
      streaming: false,
    });
    teardown();   // flips isStreaming off, clears Rust state
  });

  cmd.on("close", () => {
    teardown();
    if (!dispatch.resultReceived) {
      // Stream ended without a result event — surface a system bubble per
      // RESEARCH spike landmine #3 + AI-SPEC §4b
      dispatch.messages.push({
        id: uid(),
        role: "system",
        text: "stream ended unexpectedly (no result event)",
        streaming: false,
      });
      return;
    }
    // Finalize-render path: walk all .assistant bubbles with streaming===false
    // and replace their textContent with sanitizeMarkdown(...) HTML, then
    // renderKatexInDom over each.
    requestAnimationFrame(() => {
      const els = document.querySelectorAll<HTMLElement>('[data-msg-role="assistant"][data-msg-streaming="false"]');
      for (const el of Array.from(els)) {
        const raw = el.textContent ?? "";
        const sanitized = sanitizeMarkdown(raw);
        // The sanitized HTML has been DOMPurify-cleaned at the source; safe
        // to inject via Range#createContextualFragment which doesn't execute
        // scripts but parses DOM correctly.
        const range = document.createRange();
        range.selectNodeContents(el);
        range.deleteContents();
        const frag = range.createContextualFragment(sanitized);
        el.appendChild(frag);
        renderKatexInDom(el);
      }
    });
  });

  // 4. Spawn — and HAND THE PID TO RUST (plan 01-04 IPC contract)
  const child = await cmd.spawn();
  await invoke("register_session_pid", { pid: child.pid });

  // TOKENICODE finalizeOnce pattern — guard against double-cleanup
  finalizeOnce = () => {
    finalizeOnce = null;
    cmdRef = null;
    dispatch.isStreaming = false;
    pulseDotVisible = false;
  };
}

function teardown() {
  if (finalizeOnce) finalizeOnce();
  // Tell Rust to release the registry slot (natural close path).
  // Best-effort — if invoke fails (Tauri runtime gone), the kill_pgid hook
  // catches the next ExitRequested.
  invoke("clear_session_pid").catch(() => {});
}

async function onStop() {
  if (!dispatch.isStreaming) return;
  // D-19 — Rust state machine kills via stop_session command. Already-streamed
  // text in dispatch.messages is preserved (we don't truncate the buffer).
  try {
    await invoke("stop_session");
  } catch (e) {
    console.warn("[stop_session] invoke failed", e);
  }
  // teardown will fire from cmd.on("close") which the SIGKILL forces
}
```

**Hotkey unbinding** (REQ-6 acceptance L145) — install a window-level keydown listener that intercepts the unbound combos:
```
const UNBOUND = ["KeyL", "KeyK", "Comma", "KeyP", "KeyO", "KeyN", "KeyR"];
//             Cmd+L, Cmd+K, Cmd+,, Cmd+P, Cmd+O, Cmd+N, Cmd+R
// Plus Cmd+Shift+P (KeyP + shiftKey)
function onWindowKeydown(e: KeyboardEvent) {
  if (!(e.metaKey || e.ctrlKey)) return;
  if (UNBOUND.includes(e.code)) {
    e.preventDefault();
    // Silent — REQ-6 acceptance: NO UI response, NO error log
  }
}
```

Cmd+Q + Cmd+W are NOT in this list — they're handled by Tauri's WindowEvent::CloseRequested + RunEvent::ExitRequested hooks (plan 01-04). The browser-level keydown for Cmd+Q on macOS doesn't reach the WebView (system intercepts), but Cmd+W does — Tauri's WindowEvent::CloseRequested fires for Cmd+W (a single-window app), so plan 01-04's hook union catches it. This plan does NOT explicitly bind or unbind Cmd+Q / Cmd+W in JS.

The finalize-on-close walker greps DOM for `[data-msg-role="assistant"][data-msg-streaming="false"]` to find finalized bubbles. For a finalized bubble, `msg.text` is the raw streamed plain text; the walker replaces the text node with sanitized HTML (DOMPurify-cleaned at source via `sanitizeMarkdown`).

For `msg.text` containing literal `<` or `>` from the streaming path (chunky monospace `<pre>` displays raw text via Svelte's auto-text-escape), the bubble is safe pre-finalization. After finalization, sanitizeMarkdown handles all escaping. The user-bubble + tool-use chrome rely on Svelte's auto-escape (no `{@html}`).

**ONLY ONE `{@html}` site in this plan:** the system bubble error variant, where `msg.text` is already `escapeHtml`-escaped at dispatch time (see plan 01-03's stream-dispatch.ts — system error path). Anywhere else uses Svelte's auto-text-escape.

**Native traffic-light avoidance:** Splitter (plan 01-05) already applies `padding-top: 36px` on the right pane. This component does NOT add additional top padding to compensate. The chat scroller starts immediately under that 36px reserved zone.

**System bubble error variant test path:** subprocess error (e.g., `claude` not on PATH) → spike-findings says raw error message surfaces. This component pushes a system bubble via the dispatch path which `escapeHtml`s the message before it lands in `msg.text`; the bubble renders via `{@html msg.text}` (the only legal `{@html}` site, because the text was sanitized at dispatch). The visual treatment uses `var(--error)` left-border + text color.

          // execution. The bytes inserted are DOMPurify-clean (sanitize.ts gate).
          const range = document.createRange();
          range.selectNodeContents(el);
          range.deleteContents();
          const frag = range.createContextualFragment(sanitized);
          el.appendChild(frag);
          renderKatexInDom(el);
          el.dataset.msgFinalized = "true";   // idempotency flag
        }
      });
    });

    // Spawn — hand PID to Rust state machine (plan 01-04).
    const child = await cmd.spawn();
    await invoke("register_session_pid", { pid: child.pid });

    finalizeOnce = () => {
      finalizeOnce = null;
      dispatch.isStreaming = false;
      pulseDotVisible = false;
    };
  }

  function teardown() {
    if (finalizeOnce) finalizeOnce();
    // Best-effort release the Rust registry slot.
    invoke("clear_session_pid").catch(() => {});
  }

  async function onStop() {
    if (!dispatch.isStreaming) return;
    try {
      await invoke("stop_session");
    } catch (e) {
      console.warn("[stop_session] invoke failed", e);
    }
    // cmd.on("close") fires from the SIGKILL → teardown runs there.
  }

  // ---------- keyboard ----------
  function handleKey(e: KeyboardEvent) {
    if (e.key === "Enter" && !e.shiftKey && !e.isComposing) {
      e.preventDefault();
      sendPrompt();
    }
    // Shift+Enter falls through to default textarea newline (D-20).
  }

  // Hotkey unbinding — REQ-6 L145 acceptance.
  // Codes for the unbound set; Cmd+Shift+P shares Code "KeyP" with Cmd+P.
  const UNBOUND_CODES = new Set([
    "KeyL", "KeyK", "Comma", "KeyP", "KeyO", "KeyN", "KeyR",
  ]);
  function onWindowKeydown(e: KeyboardEvent) {
    if (!(e.metaKey || e.ctrlKey)) return;
    if (UNBOUND_CODES.has(e.code)) {
      e.preventDefault();
      // Silent — NO UI response, NO error log per REQ-6 acceptance.
    }
  }

  // ---------- scroll ----------
  function scrollToBottomMaybe() {
    if (!scroller) return;
    const dist = scroller.scrollHeight - scroller.scrollTop - scroller.clientHeight;
    if (dist < 100) {
      // Auto-stick to bottom only if user hasn't scrolled up to read past content.
      requestAnimationFrame(() => {
        scroller.scrollTop = scroller.scrollHeight;
      });
    }
  }

  // ---------- lifecycle ----------
  onMount(() => {
    window.addEventListener("keydown", onWindowKeydown);
    if (inputBox) inputBox.focus();
  });
  onDestroy(() => {
    window.removeEventListener("keydown", onWindowKeydown);
    teardown();   // best-effort cleanup if component unmounts mid-stream
  });

  // ---------- helpers ----------
  let _uidCounter = 0;
  function uid(): string {
    return `m_${Date.now()}_${_uidCounter++}`;
  }

  function isErrorMsg(m: Msg): boolean {
    // Heuristic: system bubbles whose text begins with "stream ended" OR
    // contains escaped error markers. Plan 01-03's dispatch pushes errors
    // with a distinct shape we could detect; for now, any system msg with
    // text content uses --error styling if it begins with the
    // "stream ended" sentinel OR contains "&lt;" (escapeHtml output).
    return m.text.startsWith("stream ended") || m.text.includes("&lt;");
  }
</script>
